> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 电子版 PDF、Word、拍照（可多张 jpg/png）和扫描 PDF 都从现在这个提交口进入同一条拆题；没字的页看图最多 3 次，读不到的页不编题。

**Tech Stack:** Python 3.11、FastAPI、PyMuPDF、pytest、React 18。

```python
"""Decide when a page needs vision and retry that page up to three times."""

from __future__ import annotations

from collections.abc import Callable

MAX_VISION_ATTEMPTS = 3
EMPTY_PAPER_MESSAGE = "这些页无法识别，请换清晰件再交。"


def page_text_is_sparse(text: str) -> bool:
    blob = text or ""
    cjk = sum(1 for ch in blob if "\u4e00" <= ch <= "\u9fff")
    letters = sum(1 for ch in blob if ch.isascii() and ch.isalpha())
    return cjk < 4 and letters < 4


def read_sparse_page(read_once: Callable[[], str], *, max_attempts: int = MAX_VISION_ATTEMPTS) -> str | None:
    for _ in range(max_attempts):
        try:
            text = (read_once() or "").strip()
        except Exception:
            continue
        if not page_text_is_sparse(text):
            return text
    return None
```

```python
"""Build one exam raw_text from PDF pages and/or photo files at submit time."""

from __future__ import annotations

from collections.abc import Callable
from pathlib import Path

import fitz

from app.services.page_text import page_text_is_sparse, read_sparse_page
from app.services.parsers.docx_parser import parse_docx

IMAGE_SUFFIXES = {".jpg", ".jpeg", ".png"}
MIXED_UPLOAD_MESSAGE = "请分开交"
DOC_UNSUPPORTED = "暂不支持 .doc，请另存为 .docx 或 PDF"


def extract_exam_text(
    paths: list[Path],
    *,
    read_image_text: Callable[[bytes], str],
) -> str:
    if not paths:
        return ""
    kinds = []
    for path in paths:
        suf = path.suffix.lower()
        if suf == ".doc":
            raise ValueError(DOC_UNSUPPORTED)
        if suf in IMAGE_SUFFIXES:
            kinds.append("image")
        elif suf == ".pdf":
            kinds.append("pdf")
        elif suf == ".docx":
            kinds.append("docx")
        else:
            raise ValueError(MIXED_UPLOAD_MESSAGE)
    if "image" in kinds and ("pdf" in kinds or "docx" in kinds):
        raise ValueError(MIXED_UPLOAD_MESSAGE)
    parts: list[str] = []
    if kinds[0] == "docx":
        return parse_docx(paths[0])
    if kinds[0] == "image":
        for path in paths:
            got = read_sparse_page(lambda p=path: read_image_text(p.read_bytes()))
            if got:
                parts.append(got)
        return "\n\n".join(parts)
    return _pdf_pages(paths[0], read_image_text)


def _pdf_pages(path: Path, read_image_text: Callable[[bytes], str]) -> str:
    doc = fitz.open(path)
    parts: list[str] = []
    try:
        for page in doc:
            layer = (page.get_text("text") or "").strip()
            if not page_text_is_sparse(layer):
                parts.append(layer)
                continue
            pix = page.get_pixmap(matrix=fitz.Matrix(2, 2))
            png = pix.tobytes("png")
            got = read_sparse_page(lambda data=png: read_image_text(data))
            if got:
                parts.append(got)
    finally:
        doc.close()
    return "\n\n".join(parts)
```

"""Read visible text from one page image via DeepSeek. Tests inject a fake."""

from __future__ import annotations

import base64

from langchain_core.messages import HumanMessage, SystemMessage
from langchain_openai import ChatOpenAI

from app.config import settings


def read_image_text(data: bytes) -> str:
    key = settings.require_deepseek_api_key()
    b64 = base64.b64encode(data).decode("ascii")
    llm = ChatOpenAI(
        model=settings.deepseek_model,
        api_key=key,
        base_url=settings.deepseek_base_url,
        temperature=0,
    )
    msg = llm.invoke(
        [
            SystemMessage(
                content="你是试卷识读助手。只输出这一页上可见的文字，不要解释、不要JSON。"
                "若完全没有可识文字，输出空字符串。"
            ),
            HumanMessage(
                content=[
                    {"type": "text", "text": "请输出这一页试卷上的文字。"},
                    {
                        "type": "image_url",
                        "image_url": {"url": f"data:image/png;base64,{b64}"},
                    },
                ]
            ),
        ]
    )
    c = getattr(msg, "content", "")
    if isinstance(c, str):
        return c.strip()
    if isinstance(c, list):
        parts = []
        for block in c:
            if isinstance(block, dict) and block.get("type") == "text":
                parts.append(str(block.get("text", "")))
            elif isinstance(block, str):
                parts.append(block)
        return "".join(parts).strip()
    return str(c).strip()
